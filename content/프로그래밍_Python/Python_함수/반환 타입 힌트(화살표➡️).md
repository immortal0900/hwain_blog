**화살표(`->`) = 반환 타입 힌트**
```python
def get_db() -> Database:
    #         ↑ 이 함수는 Database 타입을 반환합니다
    return MySQLDatabase()
```

## 비교
```python
# 반환 타입 없음
def get_number():
    return 42

# 반환 타입 있음
def get_number() -> int:
    #              ↑ int를 반환한다고 명시
    return 42
```

## 실제 예시
```python
# 여러 타입들
def get_name() -> str:
    return "홍길동"

def get_age() -> int:
    return 30

def get_prices() -> list:
    return [100, 200, 300]

def get_user() -> dict:
    return {"name": "김철수", "age": 25}

def process() -> None:  # 반환값 없음
    print("처리 완료")

# 실행
print(get_name())    # 홍길동
print(get_age())     # 30
print(get_prices())  # [100, 200, 300]
```
